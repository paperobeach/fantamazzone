---statistiche live per STAGIONE, GIORNATA, PARTITA

SELECT A.*, B.api_nome, D.squadra_casa, D.squadra_ospite FROM `NEW_API_STATISTICHE` A
INNER JOIN NEW_API_MAPPING_GIOCATORI B ON A.stagione = B.stagione AND A.api_player_id = B.api_player_id
INNER JOIN NEW_API_PARTITE C ON A.stagione = C.stagione AND A.api_fixture_id = C.api_fixture_id
INNER JOIN NEW_CALENDARIO_SERIE_A D ON D.stagione = C.stagione AND D.giornata = C.giornata AND D.n_partita = C.n_partita
WHERE A.STAGIONE = 2026
 AND D.giornata = 9
 AND UPPER(D.squadra_casa) = 'SASSUOLO'
 and UPPER(D.squadra_ospite) = 'LAZIO';
 
 
 ---raccordo storico giocatori
 SELECT A.*, B.*, C.descrizione AS NOME_STAGIONI_OLD, D.nome AS NOME_CORRENTE_IN_ROSA, E.descrizione AS NOME_CORRENTE_NO_ROSA FROM NEW_GIOCATORI_ANAGRAFICA A 
INNER JOIN NEW_GIOCATORI_RACCORDO B ON B.id_anagrafica = A.id_anagrafica
LEFT JOIN NEW_GIOCATORI C ON C.id = B.id_giocatore AND C.stagione = B.stagione
LEFT JOIN NEW_GIOCATORI_BASE_ASTA D ON D.id_giocatore = B.id_giocatore AND D.stagione = B.stagione
LEFT JOIN NEW_GIOCATORI_SVINCOLATI E ON E.id = B.id_giocatore AND E.stagione = B.stagione
WHERE A.ID_ANAGRAFICA = 3406;
